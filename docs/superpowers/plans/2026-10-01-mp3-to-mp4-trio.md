```typescript
  '16:9': { '720p': { w: 1280, h: 720  }, '1080p': { w: 1920, h: 1080 } },
  '9:16': { '720p': { w: 720,  h: 1280 }, '1080p': { w: 1080, h: 1920 } },
  '1:1':  { '720p': { w: 720,  h: 720  }, '1080p': { w: 1080, h: 1080 } },
  '4:5':  { '720p': { w: 864,  h: 1080 }, '1080p': { w: 1296, h: 1620 } },
}

export function resolveDimensions(aspect: string, resolution: string): { w: number; h: number } {
  return TABLE[aspect]?.[resolution] ?? TABLE['16:9']['720p']
```

```typescript
const ZERO = '00:00:00'
const norm = (v: string | undefined) => (typeof v === 'string' && v.length > 0 ? v : ZERO)

export function buildTrimArgs(start: string | undefined, end: string | undefined): string[] {
  const s = norm(start)
  const e = norm(end)
  const args: string[] = []
  if (s !== ZERO) args.push('-ss', s)
  if (e !== ZERO) args.push('-to', e)
  return args
}
```

```typescript
export type BgType = 'black' | 'color' | 'image'
export type Waveform = 'none' | 'bar' | 'line'

export interface FilterSpec {
  w: number
  h: number
  bgType: BgType
  bgColor: string            // hex, used only when bgType === 'color'
  waveform: Waveform
  captions: boolean
  captionAssName: string | null  // required when captions = true
}

export interface FilterOutput {
  /** The -filter_complex value. When no complex filter is needed, caller may fall back to -vf. */
  filter: string
  /** Stream label to pass to -map for video (e.g. "[v]" or "[0:v]"). */
  vMap: string
  /** Stream label to pass to -map for audio. */
  aMap: string
  /** True when caller must use -filter_complex (not -vf). */
  useComplex: boolean
  /** Index of the audio input (depends on whether image is input 0). */
  audioInputIndex: number
}

export function buildFilterComplex(spec: FilterSpec): FilterOutput {
  const { w, h, bgType, bgColor, waveform, captions, captionAssName } = spec
  const size = `${w}x${h}`
  const imageIsInput = bgType === 'image'
  const audioIdx = imageIsInput ? 1 : 1 // audio is always input 1 (background is lavfi or image input 0)
  const aMap = `${audioIdx}:a`
  const mode = waveform === 'bar' ? 'p2p' : 'line'
  const waveY = captions ? '0' : null // top when captions, otherwise natural center via overlay default

  // Build the background node → `[bg]`
  let parts: string[] = []
  if (bgType === 'image') {
    // Blur-fill: cover-scaled & cropped & blurred backdrop + contained foreground centered on top
    parts.push(
      `[0:v]split=2[bgsrc][fgsrc]`,
      `[bgsrc]scale=${w}:${h}:force_original_aspect_ratio=increase,crop=${w}:${h},boxblur=20:5,setsar=1[bgblur]`,
      `[fgsrc]scale=${w}:${h}:force_original_aspect_ratio=decrease,setsar=1[fg]`,
      `[bgblur][fg]overlay=(W-w)/2:(H-h)/2[bg]`,
    )
  } else {
    // The caller supplies the lavfi color as input 0 (`-f lavfi -i color=...`).
    // Here we just alias [0:v] → [bg] via the null filter.
    parts.push(`[0:v]null[bg]`)
  }

  // Add waveform overlay if requested
  let afterWaves = '[bg]'
  if (waveform !== 'none') {
    parts.push(`[${audioIdx}:a]showwaves=s=${size}:mode=${mode}:colors=white:scale=sqrt[waves]`)
    const overlayArgs = waveY !== null ? `overlay=0:${waveY}` : `overlay`
    parts.push(`[bg][waves]${overlayArgs}[waved]`)
    afterWaves = '[waved]'
  }

  // Burn captions as the final step
  if (captions && captionAssName) {
    parts.push(`${afterWaves}ass='${captionAssName}'[v]`)
    afterWaves = '[v]'
  } else if (afterWaves !== '[v]') {
    // Rename final node to [v] so downstream mapping is uniform
    parts.push(`${afterWaves}null[v]`)
    afterWaves = '[v]'
  }

  const useComplex = true
  return {
    filter: parts.join(';'),
    vMap: afterWaves,
    aMap,
    useComplex,
    audioInputIndex: audioIdx,
  }
}
```

```typescript
export async function mp3ToMp4(
  files: File[],
  options: ToolOptions,
  onProgress?: (fileIndex: number, pct: number) => void
): Promise<ConversionResult[]> {
  const results: ConversionResult[] = []

  const bgType     = (options.bgType     as 'black' | 'color' | 'image') ?? 'black'
  const bgColor    = (options.bgColor    as string) ?? '#1a1a2e'
  const bgImage    = (options.bgImage    as File | null) ?? null
  const waveform   = (options.waveform   as 'none' | 'bar' | 'line') ?? 'none'
  const aspect     = (options.aspect     as string) ?? '16:9'
  const resolution = (options.resolution as string) ?? '720p'
  const trimStart  = (options.trimStart  as string) ?? '00:00:00'
  const trimEnd    = (options.trimEnd    as string) ?? '00:00:00'
  const captions   = (options.captions   as boolean) ?? false

  const { w, h } = resolveDimensions(aspect, resolution)
  const trimArgs = buildTrimArgs(trimStart, trimEnd)

  for (let i = 0; i < files.length; i++) {
    onProgress?.(i, 2)

    try {
      const ffmpeg = await getFFmpeg()
      const file = files[i]
      const ext = file.name.split('.').pop() ?? 'mp3'
      const inputName = `audio_${i}.${ext}`
      const outputName = `out_${i}.mp4`
      const imageExt = bgImage?.name.split('.').pop() ?? 'jpg'
      const imageName = `bg_${i}.${imageExt}`
      let captionAssName: string | null = null

      // Captions pass (before encode so we know the ass file name)
      if (captions) {
        onProgress?.(i, 5)
        const transcript = await transcribeToWords(
          file,
          'fast',          // whisper-tiny.en
          'en',            // English only
          (phase, pct) => {
            // Map model/transcribe → 5..40
            const base = phase === 'extract' ? 5 : phase === 'model' ? 10 : 20
            const span = phase === 'extract' ? 5 : phase === 'model' ? 10 : 20
            onProgress?.(i, Math.round(base + (pct / 100) * span))
          },
        )
        const assText = buildASS(
          transcript.words,
          { ...DEFAULT_CAPTION_OPTIONS, styleId: 'classic', position: 'bottom' },
          'Arial',
          w,
          h,
        )
        const assBytes = new TextEncoder().encode(assText).buffer
        const assFile = await materializeCaptionFile(
          captionFileFromBytes(assBytes, { name: `subs_${i}.ass`, type: 'text/x-ass' })
        )
        captionAssName = assFile.name
        await ffmpeg.writeFile(captionAssName, new Uint8Array(await assFile.arrayBuffer()))
      }

      await ffmpeg.writeFile(inputName, await fetchFile(file))
      if (bgImage) await ffmpeg.writeFile(imageName, await fetchFile(bgImage))

      const { filter, vMap, aMap } = buildFilterComplex({
        w, h, bgType, bgColor, waveform, captions, captionAssName,
      })

      const inputArgs: string[] =
        bgType === 'image' && bgImage
          ? ['-loop', '1', '-i', imageName, ...trimArgs, '-i', inputName]
          : [
              '-f', 'lavfi',
              '-i', `color=c=${(bgType === 'black' ? '#000000' : bgColor).replace('#', '0x')}:size=${w}x${h}:rate=${waveform === 'none' ? '1' : '25'}`,
              ...trimArgs,
              '-i', inputName,
            ]

      const baseCodecArgs = ['-c:v', 'libx264', '-crf', '28', '-preset', 'ultrafast',
                             '-c:a', 'aac', '-b:a', '192k', '-shortest']

      const progressHandler = ({ progress }: { progress: number }) => {
        const base = captions ? 40 : 15
        const span = captions ? 55 : 80
        onProgress?.(i, Math.round(base + progress * span))
      }
      ffmpeg.on('progress', progressHandler)

      let data: Uint8Array<ArrayBuffer> | undefined
      try {
        await ffmpeg.exec([
          ...inputArgs,
          '-filter_complex', filter,
          '-map', vMap, '-map', aMap,
          ...baseCodecArgs,
          outputName,
        ])
        data = await ffmpeg.readFile(outputName) as Uint8Array<ArrayBuffer>
      } finally {
        ffmpeg.off('progress', progressHandler)
        await ffmpeg.deleteFile(inputName).catch(() => {})
        await ffmpeg.deleteFile(outputName).catch(() => {})
        if (bgImage) await ffmpeg.deleteFile(imageName).catch(() => {})
        if (captionAssName) await ffmpeg.deleteFile(captionAssName).catch(() => {})
      }

      if (!data || data.byteLength === 0) throw new Error('Conversion produced no output')
      const baseName = file.name.replace(/\.[^.]+$/, '')
      results.push(new File([data], `${baseName}.mp4`, { type: 'video/mp4' }))
      onProgress?.(i, 100)
    } catch (err) {
      results.push(toError(err))
    }
  }

  return results
}
```
